local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local HardDrag = {}
HardDrag.Distance = 7
HardDrag.Noclip = false
local DragConns = {}
local DragAlive = false

function HardDrag.Off()
    DragAlive = false
    for i, c in DragConns do c:Disconnect() end
    DragConns = {}
    local gui = Players.LocalPlayer.PlayerGui:FindFirstChild("ItemDraggingGUI")
    local env = gui and gui:FindFirstChild("Dragger") and getsenv(gui.Dragger)
    if env then
        for i, name in {"click", "clickEnded", "rotate", "holdDistanceChanged"} do
            pcall(restorefunction, env[name])
        end
        local Dragger = env.initializeDrag and debug.getupvalue(env.initializeDrag, 4)
        local BP = Dragger and Dragger:FindFirstChildOfClass("BodyPosition")
        local BG = env.click and debug.getupvalue(env.click, 8)
        if BP then
            BP.P = 10000
            BP.D = 800
            BP.MaxForce = Vector3.new(17000, 17000, 17000)
        end
        if typeof(BG) == "Instance" then
            BG.MaxTorque = Vector3.new(200, 200, 200)
            BG.P = 1200
            BG.D = 140
        end
    end
end

function HardDrag.On()
    HardDrag.Off()
    local LocalPlayer = Players.LocalPlayer
    local OrigScript = LocalPlayer.PlayerGui:WaitForChild("ItemDraggingGUI"):WaitForChild("Dragger")
    local env
    repeat
        task.wait()
        env = getsenv(OrigScript)
    until env and env.initializeDrag and env.click
    DragAlive = true

    local Mouse = LocalPlayer:GetMouse()
    local Camera = workspace.CurrentCamera
    local GetItem = require(ReplicatedStorage.Util.GetItem)
    local ClientIsDragging = ReplicatedStorage.Interaction.ClientIsDragging
    local Dragger = debug.getupvalue(env.initializeDrag, 4)
    local Track = debug.getupvalue(env.click, 4)
    local BP = Dragger:FindFirstChildOfClass("BodyPosition")
    local BG = debug.getupvalue(env.click, 8)

    for i, name in {"click", "clickEnded", "rotate", "holdDistanceChanged"} do
        hookfunction(env[name], function() end)
    end
    env.endDrag()

    BP.P = 500000
    BP.D = 1500
    BP.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    BG.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    BG.P = 30000
    BG.D = 300

    local holding, dragging, savedSpeed = false, false, nil
    local rotation = CFrame.new()
    local keys = {
        [Enum.KeyCode.A] = CFrame.Angles(0, 0.036, 0),
        [Enum.KeyCode.D] = CFrame.Angles(0, -0.036, 0),
        [Enum.KeyCode.W] = CFrame.Angles(-0.036, 0, 0),
        [Enum.KeyCode.S] = CFrame.Angles(0.036, 0, 0),
        [Enum.KeyCode.Q] = CFrame.Angles(0, 0, 0.036),
        [Enum.KeyCode.E] = CFrame.Angles(0, 0, -0.036),
    }

    local function drag()
        local part = Mouse.Target
        local char = LocalPlayer.Character
        if dragging or not char or not char:FindFirstChild("Head") or not env.canDrag(part) then return end
        if (Mouse.Hit.Position - char.Head.Position).Magnitude > 11 then return end

        dragging = true
        rotation = CFrame.new()
        env.initializeDrag(part, Mouse.Hit.Position)
        if Track then Track:Play(0.1, 1, 1) end

        local item = GetItem(part) or part.Parent
        local lastSend = 0
        while DragAlive and holding and env.canDrag(part) and Dragger.Parent do
            local head = char:FindFirstChild("Head")
            if not head then break end
            local target = head.Position + (Mouse.Hit.Position - head.Position).Unit * HardDrag.Distance
            local hitPart, hitPos = workspace:FindPartOnRayWithIgnoreList(Ray.new(head.Position, target - head.Position), {char, Dragger, part.Parent})
            if hitPart and not HardDrag.Noclip then target = hitPos end
            if (Camera.CFrame.Position - head.Position).Magnitude > 2 then
                target = target + Vector3.new(0, 1.8, 0)
            end

            local hum = char:FindFirstChildOfClass("Humanoid")
            local rotating = false
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then
                for key, rot in keys do
                    if UserInputService:IsKeyDown(key) then
                        rotation = rot * rotation
                        rotating = true
                    end
                end
            end
            if hum then
                if rotating and hum.WalkSpeed > 0 then
                    savedSpeed = hum.WalkSpeed
                    hum.WalkSpeed = 0
                elseif not rotating and savedSpeed then
                    hum.WalkSpeed = savedSpeed
                    savedSpeed = nil
                end
            end

            BP.Position = target
            BG.CFrame = CFrame.new(Dragger.Position, Camera.CFrame.Position) * rotation

            if tick() - lastSend > 0.25 then
                lastSend = tick()
                ClientIsDragging:FireServer(item)
            end
            RunService.Stepped:Wait()
        end

        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum and savedSpeed then
            hum.WalkSpeed = savedSpeed
            savedSpeed = nil
        end
        if Track then Track:Stop() end
        env.endDrag()
        dragging = false
    end

    table.insert(DragConns, UserInputService.InputBegan:Connect(function(input, gpe)
        if not gpe and input.UserInputType == Enum.UserInputType.MouseButton1 then
            holding = true
            task.spawn(drag)
        end
    end))
    table.insert(DragConns, UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            holding = false
        end
    end))
end

return HardDrag
